def sha(data):
    return hashlib.sha256(data).hexdigest()


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as f:
        data = value if isinstance(value, bytes) else (json.dumps(value, indent=2) + "\n").encode()
        f.write(data); f.flush(); os.fsync(f.fileno())


def relative_path(name):
    p = PurePosixPath(name)
    if p.is_absolute() or not p.parts or any(x in {"", ".", ".."} for x in p.parts):
        raise ValueError("unsafe relative input path")
    if any(x.startswith(".") for x in p.parts) and p.as_posix() != STAGING_WORKFLOW:
        raise ValueError("hidden input outside exact candidate workflow")
    if any(x in DENIED_PARTS for x in p.parts) or p.name in {"SUBMISSION.md", "AGENT_BOARD.md", "EXPERIMENTS.md"}:
        raise ValueError("excluded input class")
    if p.suffix not in EXTENSIONS:
        raise ValueError("unapproved extension")
    if p.suffix == ".log" and p.as_posix() not in WORKFLOW_LOGS:
        raise ValueError("workflow log is not explicitly named")
    if any(pattern.search(name.encode()) for pattern in [EMAIL, UUID, TOKEN]):
        raise ValueError("sensitive identifier in filename")
    return p.as_posix()


def read_stable(base, name):
    name = relative_path(name)
    path = base / name
    for parent in [path, *path.parents]:
        if parent == base.parent:
            break
        if parent.is_symlink():
            raise ValueError("symlink input excluded")
    before = path.stat()
    if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or before.st_size > MAX_FILE:
        raise ValueError("nonregular, hardlinked or oversized input")
    data = path.read_bytes(); after = path.stat()
    if (before.st_ino, before.st_dev, before.st_size, before.st_mtime_ns) != (after.st_ino, after.st_dev, after.st_size, after.st_mtime_ns):
        raise ValueError("input changed during read")
    data.decode("utf-8")
    return data


def log_relative(base, receipt_name, value):
    path = Path(value)
    if path.is_absolute():
        name = path.resolve().relative_to(base.resolve()).as_posix()
    elif value.startswith("research/" + base.name + "/"):
        name = value[len("research/" + base.name + "/"):]
    elif value.startswith(LOG_ROOTS):
        name = value
    else:
        name = (PurePosixPath(receipt_name).parent / value).as_posix()
    name = relative_path(name)
    if not name.startswith(LOG_ROOTS) or not name.endswith(".txt"):
        raise ValueError("completed receipt references a nonallowlisted log")
    return name


def completed_receipt(base, name):
    if not (RECEIPT_RE.fullmatch(name) or RECORD_RE.fullmatch(name)):
        raise ValueError("receipt path is outside the fixed receipt classes")
    d = json.loads(read_stable(base, name))
    if d.get("format") == "external-eth-command-v1":
        if type(d.get("exit_code")) is not int or type(d.get("timed_out")) is not bool or not closed_times(d.get("started_utc"), d.get("finished_utc")):
            raise ValueError("unfinished command receipt")
        return d, [(log_relative(base, name, d["output_path"]), d["output_sha256"], d["output_size_bytes"])]
    if "recording_complete" in d:
        if d.get("recording_complete") is not True or d.get("inputs_unchanged") is not True or type(d.get("checker_exit_code")) is not int or not closed_times(d.get("started_at"), d.get("finished_at")):
            raise ValueError("incomplete checker recording")
        return d, []
    if type(d.get("exit_code")) is int and type(d.get("timed_out")) is bool and closed_times(d.get("started_at"), d.get("finished_at")):
        return d, [(log_relative(base, name, d["log_path"]), d["log_sha256"], d["log_size"])]
    raise ValueError("unsupported completion schema; do not infer completion from a filename")


def closed_times(start, finish):
    try:
        a = datetime.datetime.fromisoformat(start.replace("Z", "+00:00"))
        b = datetime.datetime.fromisoformat(finish.replace("Z", "+00:00"))
        return a.tzinfo is not None and b.tzinfo is not None and b >= a
    except (AttributeError, TypeError, ValueError):
        return False


def completed_workflow_log(base, name):
    """Only fixed completed GitHub responses with exact original download bytes."""
    receipt = WORKFLOW_LOGS[name]
    rule = WORKFLOW_LOG_RULES[name]
    payload = read_stable(base, receipt)
    if sha(payload) != rule["receipt_sha256"]:
        raise ValueError("workflow download receipt differs from its fixed observed identity")
    d = json.loads(payload)
    if (d.get("schema") != rule["schema"] or type(d.get("exit_code")) is not int
            or d["exit_code"] != 0 or not closed_times(d.get("started_utc"), d.get("finished_utc"))):
        raise ValueError("unfinished or failed workflow-log download")
    if rule["schema"] == "eth-ci-job-log-download/1":
        if (d.get("run_id") != rule["run_id"] or d.get("job_id") != rule["job_id"]
                or d.get("body_path") != PurePosixPath(name).name):
            raise ValueError("workflow response job/path mismatch")
        digest, size = d.get("body_sha256"), d.get("body_bytes")
    else:
        expected_api = "repos/Truemachinery/eth-proof-replication/actions/jobs/" + str(rule["job_id"]) + "/logs"
        argv = d.get("argv", [])
        if (d.get("timed_out") is not False or len(argv) != 3
                or PurePosixPath(argv[0]).name != "gh" or argv[1:] != ["api", expected_api]
                or d.get("path") != "research/eth-external-replication-20261104/" + name):
            raise ValueError("terminal response command/path/completion mismatch")
        digest, size = d.get("sha256"), d.get("bytes")
    if digest != rule["raw_sha256"] or type(size) is not int or size != rule["raw_bytes"]:
        raise ValueError("workflow response identity differs from fixed observation")
    return receipt, digest, size


def make_plan(base, receipts, with_ci=False):
    items, skipped, bound_logs = {}, [], {}

    def add(name, kind, expected=None, size=None):
        try:
            data = read_stable(base, name)
            if expected is not None and (sha(data) != expected or len(data) != size):
                raise ValueError("completed receipt log hash/size mismatch")
            items[name] = {"path": name, "kind": kind, "original_sha256": sha(data), "original_size_bytes": len(data)}
        except (OSError, ValueError, UnicodeError) as error:
            skipped.append({"path": name, "reason": type(error).__name__})

    if with_ci and any(not (base / name).is_file() for name in CI_REQUIRED):
        raise ValueError("CI bundle is incomplete; await the CI author's frozen handoff")
    catalog = REPORTS | SOURCES | MANIFESTS
    if not with_ci:
        catalog -= CI_ALL | STAGING_INPUTS
    for name in sorted(catalog):
        add(name, "source" if name in SOURCES else "report" if name in REPORTS else "manifest")
    # Only closed checkpoint runs in this immutable manifest authorize their raw logs.
    tool_name = "verification-tools/tool-manifest-r1.json"
    if tool_name in items:
        d = json.loads(read_stable(base, tool_name))
        for run in d.get("build_and_acquisition_runs", []):
            if (type(run.get("exit_code")) is int and type(run.get("timed_out")) is bool
                    and closed_times(run.get("started_at"), run.get("finished_at"))):
                name = log_relative(base, tool_name, run["log_path"])
                add(name, "completed_log", run["log_sha256"], run["log_size"])
                if name in items:
                    bound_logs[name] = {"receipt": tool_name, "exit_code": run["exit_code"], "timed_out": run["timed_out"]}
    for name in WORKFLOW_LOGS:
        try:
            receipt, expected, size = completed_workflow_log(base, name)
            add(name, "completed_log", expected, size)
            if name in items:
                bound_logs[name] = {"receipt": receipt, "exit_code": 0, "timed_out": False,
                                    "scope": "Completed GitHub job-log response; display derivative, not subprocess raw bytes"}
        except (OSError, ValueError, UnicodeError) as error:
            skipped.append({"path": name, "reason": type(error).__name__})
    for name in receipts:
        d, logs = completed_receipt(base, name)
        add(name, "completed_receipt")
        for log, expected, size in logs:
            add(log, "completed_log", expected, size)
            if log in items:
                bound_logs[log] = {"receipt": name, "exit_code": d["exit_code"], "timed_out": d["timed_out"]}
    return {"schema": SCHEMA, "created_at": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "verdict": "No positive proof verdict inferred by exporter", "items": list(items.values()),
            "completed_log_bindings": bound_logs, "receipt_paths": receipts, "skipped": skipped, "with_ci": with_ci,
            "review": "Inspect this frozen selection before export. Export rechecks identities and allowlists."}


def transform(data, base, data_derivative=False):
    actions = []

    def replace_spans(rule, spans):
        nonlocal data
        if not spans:
            return
        before = data; out = []; cursor = 0; positions = []
        for start, end, replacement in sorted(spans):
            if start < cursor:
                continue
            out += [before[cursor:start], replacement]
            positions.append({"input_byte_start": start, "input_byte_end": end,
                              "replacement": replacement.decode("utf-8")})
            cursor = end
        out.append(before[cursor:]); data = b"".join(out)
        actions.append({"rule": rule, "input_sha256": sha(before), "output_sha256": sha(data), "replacements": positions})

    # Decode JSON values at sensitive keys without dumping the original values.
    text = data.decode("utf-8"); spans = []
    for m in re.finditer(r'"([^"\\]+)"\s*:\s*', text):
        if m.group(1) == "PATH" or SECRET_KEY.fullmatch(m.group(1)):
            try:
                _, end = json.JSONDecoder().raw_decode(text, m.end())
                spans.append((len(text[:m.end()].encode()), len(text[:end].encode()), b'"<REDACTED FIELD>"'))
            except ValueError:
                pass
    replace_spans("remove_sensitive_json_fields_including_full_PATH", spans)
    if data_derivative:
        # Logs/receipts can embed environment argv in JSON strings or Python repr.
        # Never run this rule on executable source, source views or AST literals.
        quoted_path = re.compile(rb"(?<=[\"'])PATH=[^\"'\r\n]*")
        replace_spans("full_PATH_quoted_argv_data_only", [
            (m.start(), m.end(), b"PATH=<PATH FIELD REDACTED>")
            for m in quoted_path.finditer(data) if b":/" in m.group(0)
        ])
    for rule, pattern, replacement in [
        ("local_replication_prefix", re.compile(re.escape(str(base.resolve()).encode())), b"<REPLICATION_ROOT>"),
        ("local_private_project_prefix", re.compile(re.escape(str(base.resolve().parents[1]).encode())), b"<PRIVATE_PROJECT_ROOT>"),
        ("home_prefix", HOME, b"<HOME>"), ("temporary_os_path", TEMP, b"<TEMP_PATH>"),
        ("email", EMAIL, b"<EMAIL REDACTED>"), ("uuid", UUID, b"<UUID REDACTED>"),
        ("credential_token", TOKEN, b"<TOKEN REDACTED>"),
        ("full_PATH_plaintext", re.compile(rb"(?m)^\s*(?:export\s+)?PATH\s*[:=][^\r\n]*"), b"<PATH FIELD REDACTED>"),
        ("sensitive_plaintext_field", re.compile(rb"(?im)^\s*(?:HOSTNAME|COMPUTER[ _-]*NAME|(?:[A-Z_]*(?:TOKEN|PASSWORD|PASSPHRASE|SECRET|SERIAL[ _-]*NUMBER|HARDWARE[ _-]*UUID)))\s*[:=][^\r\n]*"), b"<SENSITIVE FIELD REDACTED>"),
    ]:
        if rule == "full_PATH_plaintext" and not data_derivative:
            continue  # A Python PATH=expression is executable code, not a log dump.
        matches = [(m.start(), m.end(), replacement) for m in pattern.finditer(data)
                   if not (rule == "sensitive_plaintext_field" and PUBLIC_CREDENTIAL_REFERENCE.fullmatch(m.group(0)))]
        replace_spans(rule, matches)
    return data, actions


def portable_python(original, relative, base):
    """Explicit derived source, never represented as the executed source bytes."""
    tree = ast.parse(original.decode("utf-8")); changed = False
    name = "__eth_public_expand_path"
    if any(isinstance(n, ast.Name) and n.id == name for n in ast.walk(tree)):
        raise ValueError("portable helper name collision")

    class Paths(ast.NodeTransformer):
        def visit_Dict(self, node):
            nonlocal changed
            for index, key in enumerate(node.keys):
                value = node.values[index]
                if isinstance(key, ast.Constant) and key.value == "PATH" and isinstance(value, ast.Constant) and isinstance(value.value, str):
                    node.values[index] = ast.parse("__import__('os').environ.get('PATH', '')", mode="eval").body
                    changed = True
            return self.generic_visit(node)

        def visit_Constant(self, node):
            nonlocal changed
            if not isinstance(node.value, str):
                return node
            public, changes = transform(node.value.encode(), base)
            if not changes:
                return node
            changed = True
            value = public.decode()
            if any(t in value for t in ["<REPLICATION_ROOT>", "<HOME>", "<PRIVATE_PROJECT_ROOT>"]):
                return ast.copy_location(ast.Call(func=ast.Name(id=name, ctx=ast.Load()), args=[ast.Constant(value=value)], keywords=[]), node)
            return ast.copy_location(ast.Constant(value=value), node)

    tree = Paths().visit(tree)
    if not changed:
        return None
    depth = len(PurePosixPath(relative).parts) - 1
    helper = ast.parse("""def __eth_public_expand_path(value):
    p = __import__('pathlib').Path
    root = __import__('os').environ.get('ETH_REPLICATION_ROOT', str(p(__file__).resolve().parents[DEPTH]))
    project = __import__('os').environ.get('ETH_PROJECT_ROOT', '<PRIVATE_PROJECT_ROOT>')
    return value.replace('<REPLICATION_ROOT>', root).replace('<HOME>', str(p.home())).replace('<PRIVATE_PROJECT_ROOT>', project)
""".replace("DEPTH", str(depth))).body[0]
    insertion = 1 if tree.body and isinstance(tree.body[0], ast.Expr) and isinstance(tree.body[0].value, ast.Constant) and isinstance(tree.body[0].value.value, str) else 0
    while insertion < len(tree.body) and isinstance(tree.body[insertion], ast.ImportFrom) and tree.body[insertion].module == "__future__":
        insertion += 1
    tree.body.insert(insertion, helper); ast.fix_missing_locations(tree)
    shebang = "#!/usr/bin/env python3\n" if original.startswith(b"#!") else ""
    result = (shebang + "# Explicit portable derivative; original executed-source SHA is in PUBLIC_MANIFEST.json.\n" + ast.unparse(tree) + "\n").encode()
    compile(result.decode(), relative, "exec")  # syntax/placement validation only; never execute
    return result


def check_public(data):
    if any(pattern.search(data) for pattern in [EMAIL, UUID, TOKEN, HOME, TEMP]):
        raise ValueError("sensitive residual detected after transformation")


def export_plan(base, plan, release):
    if plan.get("schema") != SCHEMA or not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]{0,79}", release):
        raise ValueError("bad plan schema or release name")
    if len({item["path"] for item in plan["items"]}) != len(plan["items"]):
        raise ValueError("duplicate selected path")
    receipts = set(plan.get("receipt_paths", [])); logs = plan.get("completed_log_bindings", {})
    permitted = REPORTS | SOURCES | MANIFESTS | receipts | set(logs)
    if plan.get("with_ci") is not True:
        permitted -= CI_ALL | STAGING_INPUTS
    elif not CI_REQUIRED.issubset({item["path"] for item in plan["items"]}):
        raise ValueError("frozen CI selection omitted required files")
    for receipt in receipts:
        _, refs = completed_receipt(base, receipt)
        if any(log not in logs for log, _, _ in refs):
            raise ValueError("receipt/log selection is incomplete")
    # Revalidate all inputs before creating the destination. No live logs enter via arbitrary plan edits.
    originals = {}
    for item in plan["items"]:
        if set(item) != {"path", "kind", "original_sha256", "original_size_bytes"}:
            raise ValueError("unexpected selection fields")
        name = relative_path(item["path"])
        if name not in permitted:
            raise ValueError("plan tried to expand the fixed allowlist")
        if not re.fullmatch(r"[0-9a-f]{64}", item["original_sha256"]) or type(item["original_size_bytes"]) is not int:
            raise ValueError("invalid frozen file identity")
        expected_kind = "source" if name in SOURCES else "report" if name in REPORTS else "manifest" if name in MANIFESTS else "completed_receipt" if name in receipts else "completed_log"
        if item["kind"] != expected_kind:
            raise ValueError("selection changed the evidence class")
        if name in logs:
            binding = logs[name]
            receipt = binding["receipt"]
            if receipt == "verification-tools/tool-manifest-r1.json":
                d = json.loads(read_stable(base, receipt)); refs = [(log_relative(base, receipt, r["log_path"]), r["log_sha256"], r["log_size"]) for r in d["build_and_acquisition_runs"] if closed_times(r.get("started_at"), r.get("finished_at")) and type(r.get("exit_code")) is int and type(r.get("timed_out")) is bool]
            elif WORKFLOW_LOGS.get(name) == receipt:
                if receipt not in {item["path"] for item in plan["items"]}:
                    raise ValueError("workflow response omitted its download receipt")
                _, expected_hash, expected_size = completed_workflow_log(base, name)
                refs = [(name, expected_hash, expected_size)]
            else:
                _, refs = completed_receipt(base, receipt)
            expected = next(((h, size) for log, h, size in refs if log == name), None)
            if expected != (item["original_sha256"], item["original_size_bytes"]):
                raise ValueError("log is not bound to its completed receipt")
        data = read_stable(base, name)
        if sha(data) != item["original_sha256"] or len(data) != item["original_size_bytes"]:
            raise ValueError("frozen selection changed; create a new reviewed plan")
        originals[name] = data
    if sum(map(len, originals.values())) > MAX_TOTAL:
        raise ValueError("public package input budget exceeded")
    destination = base / "publication" / release
    if (base / "publication").is_symlink():
        raise ValueError("symlink publication directory excluded")
    destination.mkdir(parents=True, exist_ok=False)
    save(destination / "INCOMPLETE.json", {"schema": SCHEMA, "status": "exporting; not ready for publication"})
    records = []; assets = []
    for item in plan["items"]:
        name = item["path"]; raw = originals[name]
        public, actions = transform(raw, base, data_derivative=name not in SOURCES)
        if name in CI_ALL | STAGING_INPUTS and public != raw:
            raise ValueError("CI execution source requires redaction; author a portable reviewed source and freeze a new plan")
        output = "README.md" if name == "PUBLIC_README_TEMPLATE.md" else ".github/workflows/replicate.yml" if name == "ci/workflow-template.yml" else name
        kind = "byte_identical" if public == raw else "sanitized_derivative"
        view = None
        if name.endswith(".py") and public != raw:
            # Provide both a redacted source view and an explicitly derived portable version.
            check_public(public)
            view_path = "source-views/" + name + ".txt"
            save(destination / view_path, public)
            view = {"path": view_path, "sha256": sha(public), "size_bytes": len(public)}
            portable = portable_python(raw, name, base)
            if portable is None:
                raise ValueError("Python sanitization could not produce a labelled derivative")
            public = portable; kind = "explicit_portable_python_derivative"
        check_public(public)
        if name.endswith(".json"):
            json.loads(public)
        save(destination / output, public)
        executable = item["kind"] == "source" and bool((base / name).stat().st_mode & 0o111)
        (destination / output).chmod(0o755 if executable else 0o644)
        records.append({**item, "public_path": output, "public_sha256": sha(public), "public_size_bytes": len(public),
                        "public_mode": "755" if executable else "644",
                        "copy_kind": kind, "redactions": actions, "redacted_source_view": view,
                        "portable_transform": "Python AST literal transformation plus ast.unparse" if kind.startswith("explicit_portable") else None,
                        "private_ledger_dependency_in_original": b"mining_lab.checkpoint" in raw})
        if item["kind"] == "completed_receipt":
            d = json.loads(raw)
            if d.get("recording_complete") is True:
                assets.append({"receipt": name, "availability": "Export bytes retained locally; not in this package or a published asset yet",
                               "identity_source": "Reported completed-recorder receipt; packager did not rehash large export",
                               "raw_sha256": d.get("export_archive", {}).get("uncompressed_sha256"),
                               "raw_size_bytes": d.get("export_archive", {}).get("uncompressed_size_bytes"),
                               "gzip_sha256": d.get("export_archive", {}).get("gzip", {}).get("sha256"),
                               "gzip_size_bytes": d.get("export_archive", {}).get("gzip", {}).get("size_bytes")})
    # Do not silently publish inputs that changed after being copied.
    for name, original in originals.items():
        if sha(read_stable(base, name)) != sha(original):
            raise ValueError("input changed during export; incomplete directory retained")
    manifest = {"schema": SCHEMA, "release": release, "status": "prepared_for_review_not_published",
                "verification_verdict": "Not inferred; missing or unfinished stages are not passes",
                "exporter_sha256": sha(Path(__file__).read_bytes()), "plan_sha256": sha(json.dumps(plan, sort_keys=True).encode()),
                "files": records, "omitted_large_assets": assets, "skipped": plan.get("skipped", []),
                "redaction_contract": "Ordered rules record input/output hashes and exact input-byte spans. Removed private values are not repeated. Original evidence is unchanged and retained locally.",
                "excluded": ["upstream source checkouts", "caches/toolchains/binaries", "private project boards/notes/logs", "SUBMISSION.md and all email addresses", "unreceipted/live logs and JSONL ledgers", "raw proof exports"]}
    public_manifest, metadata_redactions = transform((json.dumps(manifest, indent=2) + "\n").encode(), base)
    check_public(public_manifest)
    if metadata_redactions:
        raise ValueError("generated publication manifest unexpectedly contains sensitive metadata")
    save(destination / "PUBLIC_MANIFEST.json", public_manifest)
    (destination / "INCOMPLETE.json").unlink()
    return {"release": str(destination.relative_to(base)), "files": len(records), "status": manifest["status"]}
